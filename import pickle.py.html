import pickle.py 
import pickle  # pickle
from abc import ABC, abstractmethod

# Абстракция
class Person(ABC):
    def __init__(self, name):
        self._name = name

    @abstractmethod
    def get_role(self):
        pass
# Инкапсуляция
class Book:
    def __init__(self, title, author, available=True):
        self._title = title
        self._author = author
        self._available = available

    @property
    def title(self):
        return self._title

    @property
    def author(self):
        return self._author

    @property
    def available(self):
        return self._available

    def borrow(self):
        if self._available:
            self._available = False
            return True
        return False

    def return_book(self):
        self._available = True


# Наследование
class User(Person):
    def __init__(self, name, user_id):
        super().__init__(name)
        self._user_id = user_id
        self._borrowed = []

    def get_role(self):
        return "Пользователь"

    @property
    def user_id(self):
        return self._user_id

    @property
    def borrowed_books(self):
        return self._borrowed

    def borrow_book(self, title):
        self._borrowed.append(title)

    def return_book(self, title):
        if title in self._borrowed:
            self._borrowed.remove(title)


class Librarian(Person):
    def __init__(self, name, emp_id):
        super().__init__(name)
        self._emp_id = emp_id

    def get_role(self):
        return "Библиотекарь"

    @property
    def emp_id(self):
        return self._emp_id

# Основной класс
class Library:
    def __init__(self):
        self.books = {}
        self.users = {}
        self.librarians = {}
        self.load_data()
        self.librarians["lib1"] = Librarian("Админ", "lib1")

    def add_book(self, title, author):
        if title not in self.books:
            self.books[title] = Book(title, author)
            return True
        return False

    def remove_book(self, title):
        if title in self.books and self.books[title].available:
            del self.books[title]
            return True
        return False

    def register_user(self, name, user_id):
        if user_id not in self.users:
            self.users[user_id] = User(name, user_id)
            return True
        return False

    def borrow_book(self, user_id, title):
        if user_id not in self.users or title not in self.books:
            return "Не найдено"
        if not self.books[title].available:
            return "Уже выдана"

        self.books[title].borrow()
        self.users[user_id].borrow_book(title)
        return "Успешно"

    def return_book(self, user_id, title):
        if user_id not in self.users or title not in self.books:
            return "Не найдено"
        if self.books[title].available:
            return "Уже в библиотеке"

        self.books[title].return_book()
        self.users[user_id].return_book(title)
        return "Успешно"

    def save_data(self):
        data = {
            'books': [(title, book.author, book.available)
                      for title, book in self.books.items()],
            'users': [(uid, user._name, user.borrowed_books)
                      for uid, user in self.users.items()]
        }
        with open('library_data.pkl', 'wb') as f:
            pickle.dump(self.books, f)  # Сохраняем книги
            pickle.dump(self.users, f)
    def load_data(self):
        """Загружает состояние библиотеки"""
        try:
            with open('library_data.pkl', 'rb') as f:  # 'rb' - read binary
                self.books = pickle.load(f)  # Загружаем книги
                self.users = pickle.load(f)  # Загружаем пользователей
        except FileNotFoundError:

            self.add_book("Бедная Лиза", "Карамзин")
            self.add_book("Мастер и Маргарита", "Булгаков")
            self.register_user("Инкогнито", "user1")
            self.librarians["lib1"] = Librarian("Админ", "lib1")
        except Exception as e:
            print(f"Ошибка при загрузке: {e}")
            # В случае ошибки создала тестовые данные
            self.books = {}
            self.users = {}
            self.add_book("Бедная Лиза", "Карамзин")
            self.add_book("Мастер и Маргарита", "Булгаков")
            self.register_user("Инкогнито", "user1")
            self.librarians["lib1"] = Librarian("Админ", "lib1")

# Интерфейс
class LibraryApp:
    def __init__(self):
        self.lib = Library()
        self.current_user = None
        self.current_role = None

    def run(self):
        print("Библиотечная система")
        while True:
            if not self.current_user:
                self.show_main_menu()
            else:
                if self.current_role == "librarian":
                    self.librarian_menu()
                else:
                    self.user_menu()

    def show_main_menu(self):
        print("\n1. Библиотекарь\n2. Пользователь\n3. Выход")
        choice = input("Выбор: ")

        if choice == "1":
            emp_id = input("ID сотрудника: ")
            if emp_id in self.lib.librarians:
                self.current_user = self.lib.librarians[emp_id]
                self.current_role = "librarian"
                print(f"Добро пожаловать, {self.current_user._name}!")
            else:
                print("Не найден!")

        elif choice == "2":
            print("1. Войти\n2. Регистрация")
            sub = input("Выбор: ")
            if sub == "1":
                uid = input("ID пользователя: ")
                if uid in self.lib.users:
                    self.current_user = self.lib.users[uid]
                    self.current_role = "user"
                    print(f"Добро пожаловать, {self.current_user._name}!")
                else:
                    print("Не найден!")
            else:
                print("Обратитесь к библиотекарю")

        elif choice == "3":
            self.lib.save_data()
            exit()

    def librarian_menu(self):
        # Полиморфизм через единый интерфейс
        print("\n1. Добавить книгу\n2. Удалить книгу\n3. Регистрация пользователя")
        print("4. Список пользователей\n5. Список книг\n6. Выход")

        choice = input("Выбор: ")
        actions = {
            "1": self.add_book,
            "2": self.remove_book,
            "3": self.register_user,
            "4": self.list_users,
            "5": self.list_books,
            "6": self.logout
        }

        if choice in actions:
            actions[choice]()

    def user_menu(self):
        print("\n1. Доступные книги\n2. Взять книгу\n3. Вернуть книгу")
        print("4. Мои книги\n5. Выход")

        choice = input("Выбор: ")
        actions = {
            "1": self.show_available,
            "2": self.borrow_book,
            "3": self.return_book,
            "4": self.my_books,
            "5": self.logout
        }

        if choice in actions:
            actions[choice]()

    # Методы библиотекаря
    def add_book(self):
        title = input("Название: ")
        author = input("Автор: ")
        if self.lib.add_book(title, author):
            print("Добавлено!")
        else:
            print("Уже существует!")

    def remove_book(self):
        title = input("Название: ")
        if self.lib.remove_book(title):
            print("Удалено!")
        else:
            print("Не найдено или выдана!")

    def register_user(self):
        name = input("Имя: ")
        uid = input("ID: ")
        if self.lib.register_user(name, uid):
            print("Зарегистрирован!")
        else:
            print("ID уже используется!")

    def list_users(self):
        print("\nПользователи:")
        for user in self.lib.users.values():
            books = ", ".join(user.borrowed_books) if user.borrowed_books else "нет"
            print(f"{user._name} (ID: {user.user_id}) | Книги: {books}")

    def list_books(self):
        print("\nКниги:")
        for book in self.lib.books.values():
            status = "✓" if book.available else "✗"
            print(f"{book.title} - {book.author} [{status}]")

    # Методы пользователя
    def show_available(self):
        print("\nДоступные книги:")
        available = [b for b in self.lib.books.values() if b.available]
        for book in available:
            print(f"{book.title} - {book.author}")

    def borrow_book(self):
        title = input("Название книги: ")
        result = self.lib.borrow_book(self.current_user.user_id, title)
        print(result)

    def return_book(self):
        title = input("Название книги: ")
        result = self.lib.return_book(self.current_user.user_id, title)
        print(result)

    def my_books(self):
        books = self.current_user.borrowed_books
        print("\nВаши книги:")
        if books:
            for book in books:
                print(f"- {book}")
        else:
            print("Нет книг")

    def logout(self):
        self.lib.save_data()
        self.current_user = None
        self.current_role = None
        print("Выход выполнен")
# Запуск
if __name__ == "__main__":
    app = LibraryApp()
    try:
        app.run()
    except KeyboardInterrupt:
        app.lib.save_data()
